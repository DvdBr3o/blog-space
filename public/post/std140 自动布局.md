来尝鲜 C++26 static reflection，首当其冲肯定是先把图形学里的 descriptor set layout binding 实现一下！之前手动绑定遭老罪了，还有宏洁癖不敢用黑魔法库，苦久矣！！

```cpp
#include <meta>
#include <print>
#include <vector>
#include <span>
#include <algorithm>

struct BindingSlot {
    uint32_t slot;
};

constexpr auto binding(uint32_t slot) -> BindingSlot {
    return {slot};
}

struct BindGroupLayoutEntry {
    uint32_t slot;
    uint32_t offset;
    uint32_t size;
};

struct BindGroupLayout {
    std::vector<BindGroupLayoutEntry> entries;
};

consteval auto get_std140_alignment(std::meta::info type) -> size_t {
    size_t s = std::meta::size_of(type);
    if (s <= 4) return 4;
    if (s <= 8) return 8;
    return 16; 
}

template<typename T>
auto std140() -> BindGroupLayout {
    constexpr auto ctx = std::meta::access_context::current();
    constexpr auto members = std::define_static_array(std::meta::nonstatic_data_members_of(^^T, ctx));
    constexpr auto size = members.size();

    std::vector<BindGroupLayoutEntry> entries;
    entries.reserve(size);
    
    size_t current_offset = 0;
    uint32_t next_slot = 0;

    template for (constexpr auto mem : std::define_static_array(members)) {
        constexpr auto type = type_of(mem);
        constexpr size_t alignment = get_std140_alignment(type);
        constexpr size_t size = size_of(type);

        current_offset = (current_offset + alignment - 1) & ~(alignment - 1);

        entries.emplace_back(BindGroupLayoutEntry {
            .slot = next_slot++,
            .offset = static_cast<uint32_t>(current_offset),
            .size = static_cast<uint32_t>(size),
        });

        current_offset += size;

        template for (constexpr auto anno : std::define_static_array(std::meta::annotations_of(mem))) {
            if constexpr (std::meta::type_of(anno) == ^^BindingSlot) {
                // 如果需要校验显式指定的 binding 和顺序是否一致，可以在这里做
                // constexpr auto val = std::meta::extract<BindingSlot>(anno);
            }
        }
    }

    return { .entries = std::move(entries) };
}

struct OtherAnno {};

using vec2 = std::array<float, 2>;
using vec3 = std::array<float, 3>;
using vec4 = std::array<float, 4>;
using ivec2 = std::array<int, 2>;
using ivec4 = std::array<int, 4>;
using ivec3 = std::array<int, 3>;
using dvec2 = std::array<double, 2>;
using dvec4 = std::array<double, 4>;
using dvec3 = std::array<double, 3>;


struct Global {
    float time;
    float delta_time;
    ivec2 aspect;
    float fov;
    vec3  color;
};

int main() {
    const auto global_layout = std140<Global>();
    std::println("Total Entries: {}", global_layout.entries.size());
    std::println("Slot | Offset | Size");
    for (const auto& entry : global_layout.entries) {
        std::println("{:4} | {:6} | {:4}", entry.slot, entry.offset, entry.size);
    }
}
```
